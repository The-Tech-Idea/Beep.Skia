Creating a Diagram

Start by creating a SkiaCanvas and a DrawingManager. The canvas provides the rendering surface, while the drawing manager orchestrates node placement, selection, and interaction:

Initialize the Diagram

using Beep.Skia;
using Beep.Skia.Canvas;
using Beep.Skia.Nodes;
using Beep.Skia.Connections;
using SkiaSharp;

var canvas = new SkiaCanvas(1200, 800);
canvas.BackgroundColor = SKColors.White;

var drawingManager = new DrawingManager();
drawingManager.AttachCanvas(canvas);
drawingManager.GridSize = 20;
drawingManager.SnapToGrid = true;

The GridSize and SnapToGrid properties enable precise node alignment. The canvas dimensions define the scrollable diagram area, and the drawing manager automatically handles viewport transformations, zooming, and panning.

Adding Components

Beep.Skia provides over 200 node types organized into 18 diagram families. For a flowchart, use FlowChartNode instances. Each node supports customizable shapes, labels, colors, and ports:

Add FlowChart Nodes

var startNode = new FlowChartNode("Start")
{
    Position = new SKPoint(200, 100),
    Width = 140,
    Height = 60,
    FillColor = SKColors.LightGreen,
    StrokeColor = SKColors.DarkGreen,
    Shape = FlowChartShape.RoundedRectangle
};
drawingManager.AddNode(startNode);

var inputNode = new FlowChartNode("Receive Order")
{
    Position = new SKPoint(200, 220),
    Width = 160,
    Height = 70,
    FillColor = SKColors.LightBlue,
    StrokeColor = SKColors.DarkBlue,
    Shape = FlowChartShape.Rectangle
};
drawingManager.AddNode(inputNode);

var processNode = new FlowChartNode("Validate Order")
{
    Position = new SKPoint(200, 350),
    Width = 170,
    Height = 70,
    FillColor = SKColors.LightYellow,
    StrokeColor = SKColors.DarkGoldenrod,
    Shape = FlowChartShape.Rectangle
};
drawingManager.AddNode(processNode);

var decisionNode = new FlowChartNode("Valid?")
{
    Position = new SKPoint(200, 480),
    Width = 130,
    Height = 80,
    FillColor = SKColors.LightSalmon,
    StrokeColor = SKColors.DarkRed,
    Shape = FlowChartShape.Diamond
};
drawingManager.AddNode(decisionNode);

var endNode = new FlowChartNode("End")
{
    Position = new SKPoint(200, 620),
    Width = 140,
    Height = 60,
    FillColor = SKColors.LightCoral,
    StrokeColor = SKColors.DarkRed,
    Shape = FlowChartShape.RoundedRectangle
};
drawingManager.AddNode(endNode);

Connecting Nodes

Use the Connect method to create connections between nodes. Beep.Skia automatically manages port assignment and connection routing. You can customize line styles, arrowheads, and labels:

Connect Nodes

var conn1 = drawingManager.Connect(startNode, inputNode);
conn1.LineColor = SKColors.Gray;
conn1.LineWidth = 2;
conn1.ArrowEnd = true;
conn1.Label = "Begin";

var conn2 = drawingManager.Connect(inputNode, processNode);
conn2.LineColor = SKColors.Gray;
conn2.LineWidth = 2;
conn2.ArrowEnd = true;

var conn3 = drawingManager.Connect(processNode, decisionNode);
conn3.LineColor = SKColors.Gray;
conn3.LineWidth = 2;
conn3.ArrowEnd = true;

var conn4Yes = drawingManager.Connect(decisionNode, endNode);
conn4Yes.LineColor = SKColors.Green;
conn4Yes.LineWidth = 2;
conn4Yes.ArrowEnd = true;
conn4Yes.Label = "Yes";
conn4Yes.RoutingStyle = ConnectionRouting.Orthogonal;

var errorNode = new FlowChartNode("Report Error")
{
    Position = new SKPoint(500, 500),
    Width = 150,
    Height = 60,
    FillColor = SKColors.LightPink,
    StrokeColor = SKColors.DarkRed,
    Shape = FlowChartShape.Rectangle
};
drawingManager.AddNode(errorNode);

var conn4No = drawingManager.Connect(decisionNode, errorNode);
conn4No.LineColor = SKColors.Red;
conn4No.LineWidth = 2;
conn4No.ArrowEnd = true;
conn4No.Label = "No";
conn4No.RoutingStyle = ConnectionRouting.Orthogonal;

canvas.Invalidate();

Connection routing styles include Direct, Orthogonal, Curved, and Bezier. The Orthogonal style produces clean right-angle paths suitable for flowchart diagrams.

Running the Application

To render the diagram in a desktop application, host the canvas inside a SkiaSharp view control. The complete example integrates all steps into a single runnable program:

Complete FlowChart Application

using Beep.Skia;
using Beep.Skia.Canvas;
using Beep.Skia.Nodes;
using Beep.Skia.Connections;
using SkiaSharp;
using SkiaSharp.Views.Desktop;

public class DiagramHost : Form
{
    private readonly SKControl _skControl;
    private readonly DrawingManager _drawingManager;
    private readonly SkiaCanvas _canvas;

    public DiagramHost()
    {
        Text = "Beep.Skia FlowChart Demo";
        Size = new Size(1200, 800);

        _canvas = new SkiaCanvas(3000, 2000);
        _canvas.BackgroundColor = SKColors.White;

        _drawingManager = new DrawingManager();
        _drawingManager.AttachCanvas(_canvas);
        _drawingManager.GridSize = 20;
        _drawingManager.SnapToGrid = true;

        _skControl = new SKControl
        {
            Dock = DockStyle.Fill
        };
        _skControl.PaintSurface += OnPaintSurface;
        _skControl.MouseDown += OnMouseDown;
        _skControl.MouseMove += OnMouseMove;
        _skControl.MouseUp += OnMouseUp;
        Controls.Add(_skControl);

        BuildDiagram();
    }

    private void BuildDiagram()
    {
        var start = CreateNode("Start", 200, 100, 140, 60,
            SKColors.LightGreen, SKColors.DarkGreen,
            FlowChartShape.RoundedRectangle);
        _drawingManager.AddNode(start);

        var receive = CreateNode("Receive Order", 200, 220, 160, 70,
            SKColors.LightBlue, SKColors.DarkBlue,
            FlowChartShape.Rectangle);
        _drawingManager.AddNode(receive);

        var validate = CreateNode("Validate Order", 200, 350, 170, 70,
            SKColors.LightYellow, SKColors.DarkGoldenrod,
            FlowChartShape.Rectangle);
        _drawingManager.AddNode(validate);

        var decision = CreateNode("Valid?", 200, 480, 130, 80,
            SKColors.LightSalmon, SKColors.DarkRed,
            FlowChartShape.Diamond);
        _drawingManager.AddNode(decision);

        var end = CreateNode("End", 200, 620, 140, 60,
            SKColors.LightCoral, SKColors.DarkRed,
            FlowChartShape.RoundedRectangle);
        _drawingManager.AddNode(end);

        var error = CreateNode("Report Error", 500, 500, 150, 60,
            SKColors.LightPink, SKColors.DarkRed,
            FlowChartShape.Rectangle);
        _drawingManager.AddNode(error);

        var c1 = _drawingManager.Connect(start, receive);
        c1.LineColor = SKColors.Gray; c1.LineWidth = 2; c1.ArrowEnd = true;

        var c2 = _drawingManager.Connect(receive, validate);
        c2.LineColor = SKColors.Gray; c2.LineWidth = 2; c2.ArrowEnd = true;

        var c3 = _drawingManager.Connect(validate, decision);
        c3.LineColor = SKColors.Gray; c3.LineWidth = 2; c3.ArrowEnd = true;

        var c4 = _drawingManager.Connect(decision, end);
        c4.LineColor = SKColors.Green; c4.LineWidth = 2; c4.ArrowEnd = true;
        c4.Label = "Yes"; c4.RoutingStyle = ConnectionRouting.Orthogonal;

        var c5 = _drawingManager.Connect(decision, error);
        c5.LineColor = SKColors.Red; c5.LineWidth = 2; c5.ArrowEnd = true;
        c5.Label = "No"; c5.RoutingStyle = ConnectionRouting.Orthogonal;

        _canvas.Invalidate();
    }

    private static FlowChartNode CreateNode(string label,
        float x, float y, float w, float h,
        SKColor fill, SKColor stroke, FlowChartShape shape)
    {
        return new FlowChartNode(label)
        {
            Position = new SKPoint(x, y),
            Width = w,
            Height = h,
            FillColor = fill,
            StrokeColor = stroke,
            Shape = shape
        };
    }

    private void OnPaintSurface(object sender, SKPaintSurfaceEventArgs e)
    {
        _canvas.Render(e.Surface.Canvas);
    }

    private void OnMouseDown(object sender, MouseEventArgs e)
    {
        _drawingManager.HandleMouseDown(e.X, e.Y, e.Button);
    }

    private void OnMouseMove(object sender, MouseEventArgs e)
    {
        _drawingManager.HandleMouseMove(e.X, e.Y);
    }

    private void OnMouseUp(object sender, MouseEventArgs e)
    {
        _drawingManager.HandleMouseUp(e.X, e.Y, e.Button);
    }

    [STAThread]
    public static void Main()
    {
        ApplicationConfiguration.Initialize();
        Application.Run(new DiagramHost());
    }
}

Build and run the project. You should see an interactive flowchart with drag-and-drop node repositioning, zoom, and pan capabilities. Click on nodes to select them, and use the mouse wheel to zoom in and out.
